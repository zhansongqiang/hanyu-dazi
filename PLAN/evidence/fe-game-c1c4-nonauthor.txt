### 7.2 `c1-audio-http.cjs` —— C1 的 HTTP 逐文件字节比对

```js
/* fe-game 非作者复核：content 泳道 C1 的 HTTP 逐文件检查
 *
 * 目的：C1 不只是「磁盘上有 40 个 mp3」，还要证明**接口真的把这 40 个字节原样发得出来**。
 * 做法：对**真实 app**（src/index.ts 起的服务，含真实路由与中间件）逐个 GET /audio/<point_key>.mp3，
 *       与磁盘文件比字节数与 md5，并记录 Content-Type；再做一次 404 兜底检查。
 * 不需要任何 repo 内的 import —— 服务另起，本脚本只发 HTTP。
 *
 * 用法：node c1-audio-http.cjs <baseUrl> <audioDir>
 * 退出码：0 = 40/40 全等 + 404 兜底成立；1 = 有不等；2 = 用法/环境错。
 */
const http = require('http');
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');

const base = process.argv[2];
const dir = process.argv[3];
if (!base || !dir) { console.error('用法: node c1-audio-http.cjs <baseUrl> <audioDir>'); process.exit(2); }
if (!fs.existsSync(dir)) { console.error('音频目录不存在: ' + dir); process.exit(2); }

function get(url) {
  return new Promise((resolve, reject) => {
    http.get(url, res => {
      const chunks = [];
      res.on('data', c => chunks.push(c));
      res.on('end', () => resolve({ status: res.statusCode, headers: res.headers, body: Buffer.concat(chunks) }));
    }).on('error', reject);
  });
}
const md5 = b => crypto.createHash('md5').update(b).digest('hex');

(async () => {
  const files = fs.readdirSync(dir).filter(f => f.toLowerCase().endsWith('.mp3')).sort();
  console.log('baseUrl   = ' + base);
  console.log('audioDir  = ' + dir);
  console.log('磁盘 mp3  = ' + files.length + ' 个');
  console.log('');

  let ok = 0, bad = 0;
  const ctSet = new Set();
  for (const f of files) {
    const disk = fs.readFileSync(path.join(dir, f));
    const r = await get(base + '/audio/' + encodeURIComponent(f));
    const ct = String(r.headers['content-type'] || '');
    ctSet.add(ct);
    const same = r.status === 200 && r.body.length === disk.length && md5(r.body) === md5(disk);
    if (same) { ok++; }
    else {
      bad++;
      console.log('  [FAIL] ' + f + '  HTTP ' + r.status
        + '  取回 ' + r.body.length + ' B / ' + md5(r.body)
        + '  vs 磁盘 ' + disk.length + ' B / ' + md5(disk));
    }
  }
  // 打印前 3 个样本的逐字节读数，供记录里留痕
  for (const f of files.slice(0, 3)) {
    const disk = fs.readFileSync(path.join(dir, f));
    const r = await get(base + '/audio/' + encodeURIComponent(f));
    console.log('  样本 ' + f + '：HTTP ' + r.status + '  ' + r.body.length + ' B  md5 ' + md5(r.body)
      + '  == 磁盘 ' + disk.length + ' B  md5 ' + md5(disk) + '  → ' + (md5(r.body) === md5(disk) ? '逐字节相同' : '不同'));
  }
  console.log('');
  console.log('  Content-Type 集合 = ' + JSON.stringify([...ctSet]));
  console.log('  逐文件比对：' + ok + '/' + files.length + ' 通过，' + bad + ' 不通过');

  // 404 兜底：不存在的音频必须 404，且不能回一段音频
  const missing = await get(base + '/audio/__definitely-not-here__.mp3');
  const notAudio = !/^audio\//.test(String(missing.headers['content-type'] || ''));
  console.log('  404 兜底：/audio/__definitely-not-here__.mp3 → HTTP ' + missing.status
    + '  Content-Type ' + JSON.stringify(String(missing.headers['content-type'] || ''))
    + '  ' + (missing.status === 404 && notAudio ? '成立' : '不成立'));

  const pass = bad === 0 && files.length === 40 && ok === 40
    && ctSet.size === 1 && ctSet.has('audio/mpeg')
    && missing.status === 404 && notAudio;
  console.log('');
  console.log(pass ? 'C1_HTTP_EXIT=0 全部成立' : 'C1_HTTP_EXIT=1 有断言不成立');
  process.exit(pass ? 0 : 1);
})().catch(e => { console.error('自检崩了: ' + (e && e.stack || e)); process.exit(2); });
```
